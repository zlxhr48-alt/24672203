/* --- 기존 변수 선언 아래에 추가 --- */ const GAS_URL = "여기에_배포한_웹_앱_URL을_넣으세요"; // 1단계에서 받은 URL /* --- 기존 endGame 함수를 아래와 같이 수정 --- */ function endGame() { isGameRunning = false; if(timerInterval) clearInterval(timerInterval); document.getElementById("game-screen").classList.add("hidden"); document.getElementById("result-screen").classList.remove("hidden"); document.getElementById("final-score-display").innerText = `${score} / ${selectedMissions.length}`; const resultTitle = document.getElementById("result-title"); if(score === selectedMissions.length) { resultTitle.innerText = "포즈 마스터 탄생! 🏆"; } else if (score > 0) { resultTitle.innerText = "수고하셨습니다! 🎉"; } else { resultTitle.innerText = "조금 더 연습해볼까요? 💪"; } // [추가된 기능] 게임 종료 시 구글 시트로 데이터 전송 sendGameResult(score); } /* --- 데이터 전송을 위한 신규 함수 추가 --- */ function sendGameResult(finalScore) { // 전송할 데이터 객체 const data = { timestamp: new Date().toLocaleString(), score: finalScore, totalMissions: selectedMissions.length, missionList: selectedMissions.join(", ") // 수행한 미션 목록 }; fetch(GAS_URL, { method: 'POST', mode: 'no-cors', // CORS 정책 우회 cache: 'no-cache', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify(data) }) .then(() => console.log("데이터가 성공적으로 전송되었습니다.")) .catch(err => console.error("전송 중 오류 발생:", err)); }
